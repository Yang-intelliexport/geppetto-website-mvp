```
标准公差等级：
```

```python
# 过程能力分析示例
def calculate_cpk(measurements, upper_spec, lower_spec):
    mean = np.mean(measurements)
    std = np.std(measurements, ddof=1)
    
    cpu = (upper_spec - mean) / (3 * std)
    cpl = (mean - lower_spec) / (3 * std)
    
    cpk = min(cpu, cpl)
    return cpk

# 目标：关键尺寸 Cpk > 1.33
```
